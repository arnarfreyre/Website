📊 Analytics Tables

Web Analytics Events User Behavior

Track user interactions, page views, and conversion events with session tracking.

CREATE TABLE web_events (
    -- Event Identifiers
    event_id UUID DEFAULT generateUUIDv4(),
    event_timestamp DateTime64(3),
    event_date Date DEFAULT toDate(event_timestamp),

    -- User & Session
    user_id String,
    session_id String,
    is_new_user UInt8,

    -- Event Details
    event_type LowCardinality(String), -- pageview, click, scroll, etc.
    page_url String,
    page_title String,
    referrer_url Nullable(String),

    -- User Properties
    country LowCardinality(String),
    city String,
    device_type LowCardinality(String),
    browser LowCardinality(String),
    os LowCardinality(String),

    -- Metrics
    page_load_time_ms UInt32,
    time_on_page_seconds UInt32,
    scroll_depth_percent UInt8,

    -- Custom Properties
    utm_source Nullable(String),
    utm_medium Nullable(String),
    utm_campaign Nullable(String),
    custom_properties Map(String, String)
) ENGINE = MergeTree()
PARTITION BY toYYYYMM(event_date)
ORDER BY (user_id, event_timestamp)
SAMPLE BY cityHash64(user_id)
TTL event_date + INTERVAL 365 DAY;
UUID Generation Session Tracking UTM Parameters TTL Sampling

⏰ Time Series Tables

Metrics Time Series Monitoring

High-performance time series storage for metrics and monitoring data.

CREATE TABLE metrics (
    -- Time
    timestamp DateTime64(3),
    date Date DEFAULT toDate(timestamp),

    -- Metric Identity
    metric_name LowCardinality(String),
    host LowCardinality(String),
    datacenter LowCardinality(String),

    -- Values
    value Float64,
    min_value Float64,
    max_value Float64,
    sum_value Float64,
    count UInt32,

    -- Tags for filtering
    tags Map(LowCardinality(String), String),

    -- Metadata
    unit LowCardinality(String), -- bytes, ms, percent, etc.
    aggregation_type Enum8('gauge' = 1, 'counter' = 2, 'histogram' = 3)
) ENGINE = MergeTree()
PARTITION BY toStartOfDay(date)
ORDER BY (metric_name, host, timestamp)
TTL date + INTERVAL 30 DAY DELETE,
     date + INTERVAL 7 DAY GROUP BY
        metric_name, host, datacenter, toStartOfHour(timestamp)
     SET
        value = avg(value),
        min_value = min(min_value),
        max_value = max(max_value),
        sum_value = sum(sum_value),
        count = sum(count);
Rollup TTL Map Tags Aggregations Data Retention
Sensor Data IoT Time Series

Store high-frequency sensor readings with automatic aggregation.

CREATE TABLE sensor_data (
    -- Timestamp with microsecond precision
    timestamp DateTime64(6),
    date Date DEFAULT toDate(timestamp),

    -- Device Information
    device_id FixedString(16), -- UUID as binary
    device_type LowCardinality(String),
    location_id UInt32,

    -- Measurements
    temperature Nullable(Float32),
    humidity Nullable(Float32),
    pressure Nullable(Float32),
    battery_level UInt8,

    -- Quality Indicators
    signal_strength Int8,
    error_flags UInt16,

    -- Materialized Columns for Fast Queries
    hour DateTime MATERIALIZED toStartOfHour(timestamp),
    is_anomaly UInt8 DEFAULT 0
) ENGINE = MergeTree()
PARTITION BY (device_type, toYYYYMM(date))
ORDER BY (device_id, timestamp)
SETTINGS
    index_granularity = 8192,
    min_bytes_for_wide_part = 0;

📝 Logging Tables

Application Logs Centralized Logging

Structured logging with full-text search capabilities and automatic cleanup.

CREATE TABLE app_logs (
    -- Timestamp and Partitioning
    timestamp DateTime64(3, 'UTC'),
    date Date DEFAULT toDate(timestamp),

    -- Log Source
    service_name LowCardinality(String),
    host LowCardinality(String),
    pod_name String,
    container_id String,

    -- Log Level and Category
    level Enum8('DEBUG' = 1, 'INFO' = 2, 'WARN' = 3, 'ERROR' = 4, 'FATAL' = 5),
    logger_name LowCardinality(String),
    thread_name String,

    -- Message Content
    message String CODEC(ZSTD(3)),
    stack_trace Nullable(String) CODEC(ZSTD(3)),

    -- Request Context
    trace_id Nullable(String),
    span_id Nullable(String),
    user_id Nullable(String),
    request_id Nullable(String),

    -- Structured Fields
    fields Map(String, String),

    -- Search Optimization
    message_tokens Array(String) MATERIALIZED
        splitByRegexp('[^a-zA-Z0-9]+', lower(message))
) ENGINE = MergeTree()
PARTITION BY (service_name, toYYYYMMDD(date))
ORDER BY (level, service_name, timestamp)
TTL date + toIntervalDay(level = 1 ? 1 : level = 2 ? 7 : 30)
SETTINGS
    index_granularity = 8192,
    merge_with_ttl_timeout = 86400;
Compression Tracing Support Dynamic TTL Token Search

Immutable audit log for compliance and security tracking.

CREATE TABLE audit_log (
    -- Event Metadata
    event_id UUID DEFAULT generateUUIDv4(),

    event_date Date DEFAULT toDate(event_time),

    -- Actor Information
    user_id String,
    user_email String,

    index_granularity = 8192,
    enable_mixed_granularity_parts = 1;

🌐 IoT Tables

Device Telemetry Real-time IoT

High-throughput telemetry ingestion with automatic aggregation.

CREATE TABLE device_telemetry (
    -- Time and Device
    timestamp DateTime64(3),
    device_id FixedString(16),
    device_type LowCardinality(String),

    -- Location
    latitude Float64,
    longitude Float64,
    altitude_m Float32,

    -- Telemetry Data
    metrics Map(LowCardinality(String), Float64),

    -- Device Status
    battery_percent UInt8,
    signal_strength_dbm Int8,
    firmware_version LowCardinality(String),

    -- Derived Fields
    geohash String MATERIALIZED geohashEncode(longitude, latitude, 8),
    hour DateTime MATERIALIZED toStartOfHour(timestamp)
) ENGINE = MergeTree()
PARTITION BY (device_type, toYYYYMMDD(timestamp))
ORDER BY (device_id, timestamp)
TTL timestamp + INTERVAL 7 DAY GROUP BY
    device_id, device_type, toStartOfFiveMinute(timestamp)
SET
    latitude = avg(latitude),
    longitude = avg(longitude),
    metrics = mapFromArrays(
        mapKeys(metrics),
        arrayMap(k -> avg(metrics[k]), mapKeys(metrics))
    );

Market Data Trading

High-frequency market data with OHLCV aggregations.

CREATE TABLE market_data (
    -- Time and Symbol
    timestamp DateTime64(9), -- Nanosecond precision
    symbol LowCardinality(String),
    exchange LowCardinality(String),

    -- Price Data
    bid_price Decimal64(4),
    ask_price Decimal64(4),
    last_price Decimal64(4),

    -- Volume
    bid_size UInt64,
    ask_size UInt64,
    last_size UInt64,
    cumulative_volume UInt64,

    -- OHLC (for aggregated views)
    open_price Decimal64(4),
    high_price Decimal64(4),
    low_price Decimal64(4),
    close_price Decimal64(4),
    vwap Decimal64(4), -- Volume Weighted Average Price

    -- Market Conditions
    spread Decimal64(4) MATERIALIZED ask_price - bid_price,
    mid_price Decimal64(4) MATERIALIZED (bid_price + ask_price) / 2,

    -- Trade Flags
    is_halted UInt8,
    is_auction UInt8
) ENGINE = MergeTree()
PARTITION BY (exchange, toYYYYMMDD(timestamp))
ORDER BY (symbol, timestamp)
TTL timestamp + INTERVAL 1 HOUR GROUP BY
    symbol, exchange, toStartOfMinute(timestamp)
SET
    open_price = first_value(last_price),
    high_price = max(last_price),
    low_price = min(last_price),
    close_price = last_value(last_price),
    cumulative_volume = sum(last_size);
Nanosecond Precision OHLCV Auto Aggregation Market Indicators

Common Patterns Best Practices 
-- Pattern 1: Materialized columns for common calculations
hour DateTime MATERIALIZED toStartOfHour(timestamp),
date Date MATERIALIZED toDate(timestamp),

-- Pattern 2: TTL with automatic rollup
TTL date + INTERVAL 30 DAY GROUP BY
    dimension1, dimension2, toStartOfHour(timestamp)
SET
    metric = avg(metric),
    count = sum(count),

-- Pattern 3: Efficient string storage
category LowCardinality(String), -- For low cardinality
id FixedString(16), -- For fixed-length IDs
content String CODEC(ZSTD(3)), -- For large text

-- Pattern 4: Nested data structures
items Nested(
    id UInt64,
    name String,
    value Float64
),

-- Pattern 5: Map for dynamic attributes
properties Map(String, String),
